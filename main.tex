\begin{abstract}
We study the expressive power of symmetric two-counter machines, namely
two-counter machines whose transitions
are invariant under swapping of counters.
\end{abstract}

\section{The model}

A 2-\vass $\V$ is symmetric if whenever it has a transition
$q \xstep{(x_1,x_2)} q'$
then it also has the transposed transition 
$q \xstep{(x_2,x_1)} q'$.

%
A \emph{symmetric counter machine} with two counters (\sym 2-\cm) is a 
counter machine whose \vass component (\ie, the \vass obtained by removing zero tests) is a symmetric \vass.
Thus, transitions are symmetric,
but zero tests need not be, and can distinguish the two counters.

We prove that reachability in \sym 2-\cm is decidable.
The proof consists of two independent sections:
\cref{sec:targetpump}, which proves a general lemma about \sym 2-\cm{s},
and \cref{sec:strategy,sec:proof}, which use that lemma (as a black-box)
to prove the decidability result.

\section{Terminology \& Notation}
\para{VASS}
In a \vass $\V$, \emph{path} is a sequence of transitions 
where the target state of every transition
matches the source targe state of the next one.
A \emph{run} is a path plus a source configuration whose
state matches with the first transition, such that
all transitions are fireable.
A \emph{cycle} is a path where the source state of the 
first matches the target state of the last one.
By $c; c'$ we denote the concatenation of paths $c, c'$, and  
by $c^k$ we denote the $k$-fold concatenation of a cycle $c$.
Firing of a single transition $q \xstep{(x_1,x_2)} q'$
we call a \emph{step} and
write $q(n_1, n_2) \step q'(n_1+x_1, n_2+x_2)$;
and $\step^\ast$ is the transitive closure of $\step$.

\medskip

We denote by $\mysize \V$ the size of $\V$, in unary representation,
and by $\norm \V$ the max-norm of $\V$, namely the maximal absolute value of
all numbers appearing in transitions.
Clearly, $\mysize \V \geq \norm \V$.

\para{Relations}
If $R\subseteq X \times Y$, $R' \subseteq Y \times Z$ and $A \subseteq X$, $B \subseteq Y$,
we define the following compositions:
\begin{alignat*}{3}
& R \circ R' &&= \{(x,z) \mid \exists y : (x,y) \in R \wedge (y,z) \in R'\},\\
& R \circ B  &&= \{x \in A \mid \exists y \in B : (x,y)\in R\},\\
& A \circ R  &&= \{y \in B \mid \exists x \in A : (x,y)\in R \}.
\end{alignat*}
If $X=Y$, then $R^\ast$ is $\bigcup_{n=0}^\infty \underset{\text{$n$ times}}{\underbrace{R \circ \cdots \circ R}}$.

\para{Semilinear Sets}
If $b, p_1, \dots, p_n \in \N^d$,
we define 
\[
\lin b {\{p_1,\dots,p_n\}} = 
\left\{ b + k_1 p_1 + \cdots + k_n p_n 
\mid k_1, \dots, k_n \in \N \right\},
\]
the linear set with base $b$ and periods $p_1, \dots, p_n$.
A semilinear set is a finite union of linear sets.
We say $X \subseteq Q \times \N^d$ is semilinear 
if for every $q \in Q$, 
the set 
\(\{n\in \N \mid (q,n)\in X\} \subseteq \N^d\)
is semilinear.
Similarly, $R \subseteq (Q\times \N^d)^2$ is semilinear if for every $q,q' \in Q$,
the set
\( \{(n,n') \mid ((q,n),(q',n')) \in R \} \subseteq \N^{2d}\)
is semilinear.




\section{Target pummping}\label{sec:targetpump}
\begin{lemma} [Target pumping] \label{lem:targetpump}
\vrm{I put everything in the same lemma to emphasize that all versions share the same $C,D$.}\par
Let $\V$ be a symmetric $2$-$\vass$.
One can compute numbers $C, D \in \N$ 
such that for every pair of control states $q, q'$ of $\V$ and  
$a, a'\in\N$ such that $a > C$ and $a' > a + C$,%
\[\text{if} \quad q(a,0)\step^* q'(0,a') \quad \text{then} \quad q(a,0) \step^* q'(0, a'+D),\]
and similarly for the other three symmetric cases:
\begin{alignat*}{3}
&\text{if} \quad q(0,a)\step^* q'(0,a') \quad &&\text{then}  \quad  q(0,a) \step^* q'(0, a'+D),\\
&\text{if} \quad q(0,a)\step^* q'(a',0) \quad &&\text{then}  \quad  q(0,a) \step^* q'(a'+D, 0),\\
&\text{if} \quad q(a,0)\step^* q'(a',0) \quad &&\text{then}  \quad  q(a,0) \step^* q'(a'+D,0).
\end{alignat*}
Moreover, $C$ is polynomial in $\mysize \V$ and $D$ is exponential in $\mysize \V$.
\end{lemma}
%
\begin{proof}
Consider a run $\sigma : q(a,0)\step^* q'(0,a')$ for some $a, a'\in \N$.
We modify the run to lift the target configuration to $q'(0,a'+\gamma D_\sigma)$ for any
$\gamma > 0$, for some particular $D_\sigma$ depending both on $\V$ and on $\sigma$, assuming
that both $a$ and the difference $a'-a$ are sufficiently large (namely larger
than some $C$, depending now only on $\V$).
As there will be only finitely many different numbers $D_\sigma$,
we argue then that taking the least common multiple of all $D_\sigma$ yields a
number $D$ satisfying the claim of the lemma.

By the LPS decomposition \cite{Blondin15}, we may assume \mywlog that 
$\sigma$ consists of a polynomial number of transitions $t_0, \ldots, t_\ell$
and cycles $c_1, \ldots, c_\ell$, each cycle of
polynomial length  (with respect to the size $\mysize \V$ of
unary representation of $\V$):
\[
\sigma \ \in \ s_0 \ c_1^* \ s_1 \ c_2^* \ \ldots \ c_\ell^* \ s_\ell,
\]
where the length of each cycle $c_i$ is at most $e$,
and both $\ell$ and $e$ are bounded by a polynomial of $\mysize \V$.
Let $C_0$ be the minimal value such that every cycle of length at most $e$ 
is fireable at $(C_0, C_0)$.

We now classify the cycles as follows: call a cycle $c_j$ \emph{major}
if its number of iterations $n_j$ is larger than $2 C_0$, and \emph{minor} otherwise.
Moreover, call a cycle
$c_j$ \emph{sum-increasing} if its effect $(x_j, y_j)$ satisfies $x_j + y_j > 0$.

As we're assuming that $a' - a > C$, note that the sum of counters
must increase by more by more than $C$ along $\sigma$. Thus, by setting an appropriately
large $C$, we can ensure that at least one \emph{major sum-increasing cycle}
will be present in the decomposition of $\sigma$.

In particular, the total effect of the one-off transitions $s_0, \ldots, s_\ell$ is bounded,
as each of them may increase the sum of the counters by at most $2 \norm \V$.
Likewise, each minor cycle may only increase the sum of the counters by at most
$4 C_0 e \norm \V$. Thus, it suffices to set
\[
    C \geq C_1 = (2 C_0 e_\mathrm{max} + 1)(\ell_\mathrm{max} + 1) 2\norm\V,
\]
where $e_\mathrm{max}$ and $\ell_\mathrm{max}$ are the (polynomial in $\mysize\V$) bounds
on $e$ and $\ell$ respectively.


We now focus our attention on the major sum-increasing cycles, and perform
case analysis in terms of the sign of each coordinate of their effect.
Specifically, the effect of a sum-increasing cycle can be:
vertical $(0,+)$, horizontal $(+,0)$, or one of $(+,-)$, $(+,+)$, or $(-,+)$.
We classify these into three cases.\mei{Could use a figure here as well.}

\para{Case 1: Some major sum-increasing cycle $c_j$ has vertical $(0,+)$ effect}
One extra iteration of this cycle yields the claim with $D_\sigma$ equal to its
vertical effect.

\para{Case 2: Some major sum-increasing cycle $c_j$ has effect $(-,+)$ or $(+,-)$}
Assume $(-,+)$ effect, the other case is similar.
\sla{check} \mei{I have checked}

For our argument it is enough if the $n_j$ is large enough to guarantee that 
in the middle
of $c_j^{n_j}$, namely after the half of $n_j$ of iterations of $c_j$, the 
transposed cycle $\overline{c_j}$ is also enabled.
(the transposed cycle $\overline{c_j}$ is obtained by transposing all transitions 
along $c_j$).
We observe the key claim, relying on the fact that  
the effect of $c_j$ is nonzero on both coordinates
(in particular note that we don't use the sum-increasing
assumption for this claim):
%
\begin{claim}  \label{claim:fireable}
If a cycle $c_j$ is repeated $n_j > 2C_0$ times, and $C_0 \leq i \leq n_j - C_0$,
the transposed cycle $\overline{c_j}$ is fireable after $c_j^i$
(regardless of the source configuration).
\end{claim}
%
\begin{proof}
The $n_j+1$ source-target configurations along the run $c_j^{n_j}$
form an arithmetic progression defined by
$(x_i, y_i) = (x_0 + ix, y_0 + iy)$.
%For a coordinate at which the
%effect of the cycle is positive, say $x > 0$, we obtain $x_i \geq C_0$
%for $i \geq C_0$. On the other hand, for a coordinate with a negative effect,
%say $y < 0$, we have $y_i = y_{n_j} - (n_j - i)y$, with $y_{n_j} \geq 0$, and
%thus $y_i \geq n_j - i$. Hence for $i \leq n_j - C_0$ we have $y_i \geq C_0$.
%As such, for $C_0 \leq i \leq n_j - C_0$, both coordinates are above $(C_0, C_0)$,
%which suffices to obtain that $\overline{c_j}$ is fireable.
It suffices to show that for appropriate $i$ we have $x_i, y_i \geq C_0$.
For coordinates at which the effect is positive, say $x > 0$,
we need $C_0 \leq i$; conversely for coordinates with negative effect, say $y < 0$,
we need $i \leq n_j - C_0$. Note that for $n_j > 2C_0$, we can satisfy both
inequalities.
\end{proof}
%

In consequence of \Cref{claim:fireable}, the cycle $\overline{c_j}$ is fireable
after $c_j^{C_0+1}$. Equivalently, $c_j; \overline{c_j}$ is fireable after $c_j^{C_0}$.
Denoting by $(-x, y)$ the effect of $c_j$, we have that $y > x$, as the cycle is
sum-increasing. Thus, the effect of $c_j; \overline{c_j}$, being $(y-x, y-x)$,
is positive on both coordinates, and $c_j; \overline{c_j}$ may be fired arbitrarily
many times at the midpoint of $c_j^{n_j}$, without interference with the remainder of the path.

% Let $k$ be the least common multiplicity of $m_2 - m_1$ and $m_1$.  
Now set $k:=x(y-x)$.
By iterating $x$ times the cycle $c_j; \overline{c_j}$,
we get a cycle with effect $(k,k)$, and then appending
$c_j^{y-x}$ we get the vertical effect.
Indeed, the effect of
\[
\big(c_j; \overline{c_j}\big)^{x}; c_j^{y-x}
\]
is $(0, (x+y)(y-x))$, and this cycle is still fireable in the middle of
$c_j^{n_j}$.
We thus get the claim with $D_\sigma = (x+y)(y-x)$.

\para{Case 3: All major sum-increasing cycles have $(+,0)$ or $(+,+)$ effect}
We will aim to increase the repetition count of one of the major sum-increasing
cycles, together with one other cycle occurring later in the run, in order to obtain
the vertical effect.
Let us restrict our attention to those major cycles $c_j$, whose effect $(x_j,y_j)$ satisfies $x_j\neq 0$.
Depending on whether $x_j>0$ or $x_j<0$, we call such cycles 
\emph{horizontally positive}, or \emph{horizontally negative} respectively.
We define \emph{steepness} of such a cycle $c_j$ as $y_j/x_j$ and denote it by $\varphi_j$.
Note that, while in general the steepness may be negative, by assumption it will never be negative
for a major sum-increasing cycle.

\begin{claim} \label{claim:steep}
There are some $i<j$ such that $c_i$ is a major sum-increasing cycle, $c_j$ is major and
horizontally negative, and $c_i$ is steeper than $c_j$ (i.e. $\varphi_i > \varphi_j$).
\end{claim}
%
\begin{proof}
%Define a \emph{segment} of a run as a sequence of consecutive major cycles that begins
%with a sum-increasing cycle that is the steepest among the sum-increasing major cycles
%that are part of the segment. We do not consider the minor cycles, nor the one-off
%transitions, that occur between the major cycles of a segment, to be part of said segment,
%and define the \emph{effect} of a segment to be the sum of the effects of only its major
%cycles.
%
%Thus, we split the run, and its total effect, into (a) the one-off transitions and minor cycles,
%(b) the major cycles that occur before the first sum-increasing major cycle, and
%(c) a number of segments, each starting with a steeper sum-increasing cycle than the last.
Suppose the claim doesn't hold. Let $c_{k_1}, \ldots c_{k_s}$ be the sequence
of major sum-increasing cycles of increasing steepness, i.e. let $c_{k_1}$
be the first major sum-increasing cycle of the run, and define
$c_{k_{i+1}}$ to be the first major sum-increasing cycle steeper than
$c_{k_i}$. Moreover, let $S_i \leq C_1$ be the total norm of all the one-off
transitions and minor cycles which occurred up to $c_i$ (including $c_i$
itself if it happens to be minor), and let $(X_i, Y_i)$ be the
counter values after $c_i$. Finally, denote by $\beta_i$ the steepness of
the most recent cycle among the $k_j$, i.e. $\beta_i = \varphi_i$ if
$i \in \{k_1, \ldots, k_s\}$, and $\beta_i = \beta_{i-1}$ otherwise. At the
boundary, set $\beta_0 = S_0 = 0$, and $(X_0, Y_0) = (a, 0)$.

Note that, if \Cref{claim:steep} indeed doesn't hold, we have $\beta_j \leq \varphi_j$ for any
horizontally negative major cycle $c_j$, with the possible exception of cycles $j < k_1$.
Indeed, otherwise we have $\varphi_i = \beta_j > \varphi_j$ as required.
Note also that for horizontally positive major cycles $c_i$, we have $\varphi_i \leq \beta_i$,
either by definition, for sum-increasing cycles, or trivially, for $\varphi_i \leq 0 \leq \beta_i$.

By induction on $i$, we will now prove that the configuration
will always satisfy
\begin{equation} \label{eq:steep-ind}
    Y_i \leq S_i (1 + \beta_i) + \beta_i X_i.
\end{equation}
\mei{Illustrate.}
At $i = 0$, equality trivially holds.
Suppose then that the inequality holds at a specific $i$. Denoting by $(x, y)$ the counter
values just before $c_{i+1}$, we have $y - Y_i \leq S_{i+1} - S_i$, and $X_i - x \leq S_{i+1} - S_i$.
Multiplying the latter by $\beta_i$ and rearranging, we obtain
\[
    y - Y_i \leq (S_{i+1} - S_i)(1 + \beta_i) + \beta_i (x - X_i),
\]
and thus
\[
    y \leq S_{i+1} (1 + \beta_i) + \beta_i x \leq S_{i+1} (1 + \beta_{i+1}) + \beta_{i+1} x,
\]
where the latter inequality follows because $\beta$ is non-decreasing.

If $c_{i+1}$ is a minor cycle, we can replace $(x, y)$ in the above reasoning by $(X_{i+1}, Y_{i+1})$,
obtaining the inductive hypothesis. Let us thus
focus on the case when $c_{i+1}$ is a major cycle, denoting by $(x_{i+1}, y_{i+1})$ the effect
of one of the $n_{i+1}$ iterations of the cycle.

If $x_{i+1} = 0$, we must have $y_{i+1} < 0$, as we handle vertical sum-increasing cycles in Case 1.
A $(0, -)$ cycle will however trivially preserve the desired inequality. It remains to consider
major cycles that are either horizontally positive, or horizonally negative. In both cases we have
$y_{i+1} = \varphi_{i+1} x_{i+1}$ and will endeavour to prove $\varphi_{i+1} x_{i+1} \leq \beta_{i+1} x_{i+1}$.
But as we've noted before, we have $\varphi_{i+1} \leq \beta_{i+1}$ whenever $x_{i+1} > 0$, and the
opposite inequality when $x_{i+1} < 0$, exactly as required to obtain $y_{i+1} \leq \beta_{i+1} x_{i+1}$.
Thus,
\[
    Y_{i+1} = y + n_{i+1} y_{i+1} \leq S_{i+1} (1 + \beta_{i+1}) + \beta_{i+1} (x + n_{i+1} x_{i+1})
    = S_{i+1} (1 + \beta_{i+1}) + \beta_{i+1} X_{i+1}.
\]

Hence we finish the induction and obtain \eqref{eq:steep-ind}, in particular for
the final configuration $(0, a')$:
\[
    a' \leq C_1 (1 + \varphi_\mathrm{max}) \leq C_1 (1 + \norm\V)
\]
By setting $C = C_1 (1 + \norm\V)$, we obtain a contradiction.
\end{proof}

By \Cref{claim:steep}, for some positive integers $I, J$, the sum of effects of
$c_i^I$ and $c_j^J$ is vertical $(0,+)$.
Extending $c_i^{n_i}$ with $c_i^I$ and $c_j^{n_j}$ with $c_j^J$ yields
a run ending in $q'(0,a'+D_\sigma)$, where $D_\sigma = ,,,$.
\sla{complete}

\medskip

The norm of the effect of every cycle of length at most $e$ is bounded
by $e\cdot\norm{\V}$, and hence
All values $D_\sigma$ are bounded by $2\cdot e^2\cdot \norm{\V}^2$, and hence
polynomially in $\mysize \V$.
Taking as $D$ the least common multiple of all $D_\sigma$ is enough for the
claim of the lemmma.

The constant $C$ must be large enough so that, even after forgetting of
the total effects of all transitions $t_0, \ldots, t_\ell$, the iterations of
cycles $c_1, \ldots, c_\ell$ increase the sum by the amount large enough so that
one of sum-incrasing cycles is iterated at least $2 C_0$ times (are required
by \Cref{claim:fireable}).
As $C_0$ is at most $e\cdot \norm \V$ and $\ell$ is polynomial in $\mysize \V$,
a polynomial value of $C$, \wrt $\mysize \V$, 
is sufficient to satisfy the above requirement.
\sla{complete the calculations}
\end{proof}


\section{Proof of Decidability}
\subsection{Strategy of proof}\label{sec:strategy}
As above, we use $\step$ to denote step relation of a 2-\vass, namely without zero tests.
First some notational preprocessing.
Replace every zero-testing transition 
\(t : q \xstep{\textsf{c}_i=0?} q'\)
by the longer two-step sequence
\[q \xstep{\textsf{c}_i=0?} S_i^t \xstep{(0,0)} q',\]
where $S_i^t$ is a new state, unique to the transition $t$.
Notice that the new \sym2-\cm thus obtained 
is equivalent to the original one, 
and that it has special states $S_i^t$
which can be entered only if counter $i$ is zero;
we call them zero-testing states.
The zero-testing states have the property
that they ``remember'' which counter is zero 
when the control of the machine is at them.
%into those corresponding to counter $\textsf{c}_1$ and to counter $\textsf{c}_2$,
%so we tell which counter is zero by looking at the state.


The strategy of the proof is as follows:
a run $\rho$ from $s$ to $t$ in the \sym2-\cm  ($s,t \in Q\times\N^2$)
can be decomposed at the zero-testing states
into segments which are \vass runs;
write
%\[ \rho=\rho_0 z_1 \rho_1 z_2 \dots \rho_{n-1}z_n\rho_n,\]
\[ \rho=\rho_0 \rho_1 \dots \rho_n,\]
where 
%the $z_i$ are the zero-testing states of $\rho$, and
each $\rho_i$ starts and ends at consecutive zero-testing states of $\rho$---%
we shall call the $\rho_i$ the \emph{\vass segments} of $\rho$.
%
Every $\rho_i$ is of one of the following forms:
\begin{alignat*}{2}
q(a,0) &\step^\ast q'(0,a')\\
q(0,a) &\step^\ast q'(0,a')\\
q(0,a) &\step^\ast q'(a',0)\\
q(a,0) &\step^\ast q'(a',0),
\end{alignat*}
where $q$ and $q'$ are zero-testing states;
notice that we may write just
\begin{equation} \label{eq:basictran}
q(a) \step^* q'(a')
\end{equation}
without specifying which of the four cases occurs,
since the (zero-testing) control states $q$, $q'$ uniquely determine that.
We follow the notational shorthand \eqref{eq:basictran} throughout. 
Without loss of generality, we also assume 
that both the source and target vector have at least one 0.

Every \vass segment $\rho_i \colon q(a) \step^\ast q'(a')$, 
depending on the values $a$ and $a'$,
falls into one of three categories,
%split the relation $\step^\ast$ into three subrelations,
%which represent different kinds of \vass runs in $(Q\times\N^2)$:
denoted by $\step_+$, $\step_-$, and $\step_0$;
the precise definitions will be given in \Cref{sec:proof}.
Thus the rechability relation is 
$\big(\step_0 \ \cup \ \step_- \ \cup \ \step_+ \big)^*$, 
an arbitrary composition
of the three relations arbitrarily many times.
Note that the composition assures that \emph{control states match}!

\bigskip\noindent
Using this categorization 
we split the run $\rho$ into three phases (possibly empty): 
\begin{itemize}
\item $\Phi_1$, which is a \emph{prefix} of $\rho$ and ends with the \emph{last} $\rho_i$ of the form $\step_+$,
\item $\Phi_2$, which is an \emph{infix} of $\rho$ and consists only of $\rho_i$'s of the form $\step_0^*$, and
\item $\Phi_3$, which is a \emph{suffix} of $\rho$ and starts with a $\rho_i$ of the form $\step_-$ 
(followed by other \vass segments of the forms $\step_-$ and $\step_0$).
\end{itemize}
It is obvious to deduce that the three phases are of the following form:
\begin{align*}
\Phi_1 :\ &((\step_0 \ \cup \ \step_-)^\ast \circ \step_+)^\ast,\\
\Phi_2 :\ &(\step_0)^\ast,\\
\Phi_3 :\ &(\step_- \circ \step_0^\ast)^\ast.
\end{align*}

To conclude, 
we will prove that $\Phi_2 \subseteq (Q \times \N)^2$ 
is an \emph{effectively semilinear} relation%
%\footnote{This means $\{n \in \N \colon (q,n) \in \Phi_3 \}$ is effectively semilinear, for every $q\in Q$.}%
,
and that for given source $s$ and target $t$, the sets
\begin{alignat*}{6}
\Phi_3 \circ \{t\} &\coloneq \{x &&\mid (x,t) \in \Phi_3 \hskip 0.5pt\} &&\subseteq Q \times \N, \\
\{s\} \circ \Phi_1 &\coloneq \{y &&\mid (s,y) \in \Phi_1 \} &&\subseteq Q \times \N
\end{alignat*}
are also \emph{effectively semilinear}.
Thus the reachability question, 
which is equivalent to%
%\( (s,t) \in \Phi_1\circ\Phi_2\circ\Phi_3,\) \ie 
\begin{equation}
\exists\, x,y : (s,x) \in \Phi_1 \wedge (x,y) \in \Phi_2 \wedge (y,t) \in \Phi_3,
\end{equation}
is decidable
(by semilinearity, the last formulation is expressible in Pressburger Arithmetic).

\para{Computing pumpable sets}
In the sequel, fix $D$ from \cref{lem:targetpump}. 
A set $X \subseteq Q \times \N$ is \emph{pumpable} 
if it is $D$-periodic:
for all $(q,a) \in X$, also $(q,a+D) \in X$.

We will compute an increasing chain 
of sets of configurations $X\subseteq Q\times\N$ which are pumpable.
%We represent these sets in vector form as functions
%$X:Q\to \powset{\N}$, where $X(q)$ is pumpable for every $q\in Q$, namely $a\in X(q)\implies a+D\in X(q)$.
The termination of computation is guaranteed by the following lemma:
\begin{lemma} \label{lem:stab}
    Every increasing chain of $X_1 \subseteq X_2 \subseteq \cdots$ of $D$-periodic sets of $\N$ stabilizes.
\end{lemma}

\begin{lemma} \label{lem:compback}

\begin{lemma} \label{lem:compfor}
One can compute 
the set 
$\{s\} \circ \big(\step_{0}^* \ \circ \ E \ \circ \ \step_+)^*$.
\end{lemma}
\begin{proof}
Again, The set $f(S) = S \ \circ \ \step_{0}^* \ \circ \ E \ \circ \ \step_+$ is
target-pumpable due to \eqref{eq:tpump},
irrespectively of $S$.
Computability is more involved, but relates again on iteration of the transformation
\[
S \quad \mapsto \quad f(S) \quad \cup \quad S,
\]
relying on \Cref{lem:stab}.

\vrm{To write that $f(S)$ consists of arithmetic progressions of period $D$,
and that we choose the minimal element of every arithmetic progression.}
To compute $f(S)$, one first establishes which remainders modulo $D$ appear in this set,
by applyting the transformation of \Cref{lem:compback} to the set 
$(\step_+ \ \circ \ M_k)$,
where $M_k$ contains all numbers equal to $k$ modulo $D$, and checking intersection with 
the set $(S\ \circ \ \step_0)$. 
Then, by exhaustive search one finds the smallest number in $f(S)$, 
for each remainder modulo $D$ separately.
\end{proof}
